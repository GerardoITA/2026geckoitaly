custom_button = {
	name = gecko_menu_toggle
	potential = { ai = no }
	trigger = { always = yes }
	effect = { hidden_effect = {
		if = { limit = { has_country_flag = gecko_menu_open NOT = { has_country_flag = gecko_intscore_gui_open } } clr_country_flag = gecko_menu_open }
		else = {
			clr_country_flag = gecko_intscore_gui_open
			if = { limit = { NOT = { has_country_flag = gecko_menu_initialized } } set_variable = { which = gecko_menu_page value = 0 } set_variable = { which = gecko_menu_index value = 0 } set_country_flag = gecko_menu_initialized }
			set_country_flag = gecko_menu_open
		}
	} }
	tooltip = gecko_menu_open_tt
}
