
add_age_modifier = {
	if = {
		limit = {
			current_age = $age$
		}
		add_country_modifier = {
			name = $name$
			duration = $duration$
			desc = until_end_of_$age$_desc
		}
	}
	else = {
		add_prestige_or_monarch_power = { amount = 50 }
	}
}
add_legitimacy_or_reform_progress = {
	tooltip = {
		legitimacy = $amount$
	}
	custom_tooltip = add_legitimacy_or_reform_progress_tt
	hidden_effect = {
		for = {
			amount = $amount$
			effect = "
				if = {
					limit = {
						NOT = { legitimacy = 99.9 }
					}
					add_legitimacy = 1
				}
				else = {
					change_government_reform_progress = 5
				}
			"
		}
	}
}
development_counter = {
	[[on_startup]
		if = {
			limit = {
				ROOT = {ai = no}
				NOT = { has_country_flag = has_development_counter_cflag }				
			}
			set_variable = {
				which = adm_dev_counter
				value = 0
			}
			set_variable = {
				which = dip_dev_counter
				value = 0
			}
			set_variable = {
				which = mil_dev_counter
				value = 0
			}
			set_variable = {
				which = total_dev_counter
				value = 0
			}
			set_country_flag = has_development_counter_cflag
		}
	]
	
	#All updates to variables in this section are in province scope.
	[[adm]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = adm_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
	[[dip]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = dip_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
	[[mil]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = mil_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
}

}
