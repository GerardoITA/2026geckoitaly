# Country scope. Preserve the former court option's thresholds and permanent bonus.
gecko_grant_colonial_expansion_effect = {
	if = {
		limit = {
			NOT = { has_country_modifier = free_colonist }
			OR = {
				AND = {
					is_colonial_nation = yes
					dip_tech = 10
				}
				AND = {
					is_colonial_nation = no
					dip_tech = 6
				}
			}
		}
		add_country_modifier = {
			name = "free_colonist"
			duration = -1
		}
	}
}
