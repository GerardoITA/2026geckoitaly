# Country scope: save before changing militarization tiers, then restore after regeneration.
# Transfer in chunks to retain two decimal places without applying gain modifiers.
gecko_store_prussian_militarization = {
	set_variable = { which = gecko_saved_prussian_militarization value = 0 }
	gecko_store_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 1 loss = -1 }
	gecko_store_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 0.1 loss = -0.1 }
	gecko_store_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 0.01 loss = -0.01 }
}

gecko_store_prussian_militarization_chunk = {
	while = {
		limit = {
			has_government_power = {
				mechanic_type = $mechanic$
				power_type = $power$
				value = $amount$
			}
		}
		add_government_power = {
			mechanic_type = $mechanic$
			power_type = $power$
			value = $loss$
			scaled_with_gain_modifier = no
		}
		change_variable = { which = gecko_saved_prussian_militarization value = $amount$ }
	}
}

gecko_restore_prussian_militarization = {
	set_government_power = {
		mechanic_type = $mechanic$
		power_type = $power$
		value = 0
	}
	gecko_restore_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 1 }
	gecko_restore_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 0.1 }
	gecko_restore_prussian_militarization_chunk = { mechanic = $mechanic$ power = $power$ amount = 0.01 }
	set_variable = { which = gecko_saved_prussian_militarization value = 0 }
}

gecko_restore_prussian_militarization_chunk = {
	while = {
		limit = {
			check_variable = { which = gecko_saved_prussian_militarization value = $amount$ }
		}
		add_government_power = {
			mechanic_type = $mechanic$
			power_type = $power$
			value = $amount$
			scaled_with_gain_modifier = no
		}
		subtract_variable = { which = gecko_saved_prussian_militarization value = $amount$ }
	}
}

# Country scope. Upgrades are one-way and transfer the current amount before regeneration.
gecko_upgrade_prussian_militarization_2 = {
	if = {
		limit = {
			has_dlc = "Domination"
			has_government_mechanic = prussian_militarization_mechanic_1
			NOT = { has_country_flag = enable_second_mil_level_flag }
			NOT = { has_country_flag = enable_third_mil_level_flag }
		}
		custom_tooltip = pru_access_normal_militarization_tt
		hidden_effect = {
			gecko_store_prussian_militarization = { mechanic = prussian_militarization_mechanic_1 power = prussian_militarized_society_1 }
			set_country_flag = enable_second_mil_level_flag
			regenerate_government_mechanics = yes
			gecko_restore_prussian_militarization = { mechanic = prussian_militarization_mechanic_2 power = prussian_militarized_society_2 }
		}
	}
}

gecko_upgrade_prussian_militarization_3 = {
	if = {
		limit = {
			has_dlc = "Domination"
			OR = {
				has_government_mechanic = prussian_militarization_mechanic_1
				has_government_mechanic = prussian_militarization_mechanic_2
			}
			NOT = { has_country_flag = enable_third_mil_level_flag }
		}
		custom_tooltip = pru_access_advanced_militarization_tt
		hidden_effect = {
			if = {
				limit = { has_government_mechanic = prussian_militarization_mechanic_2 }
				gecko_store_prussian_militarization = { mechanic = prussian_militarization_mechanic_2 power = prussian_militarized_society_2 }
			}
			else = {
				gecko_store_prussian_militarization = { mechanic = prussian_militarization_mechanic_1 power = prussian_militarized_society_1 }
			}
			clr_country_flag = enable_second_mil_level_flag
			set_country_flag = enable_third_mil_level_flag
			regenerate_government_mechanics = yes
			gecko_restore_prussian_militarization = { mechanic = prussian_militarization_mechanic_3 power = prussian_militarized_society_3 }
		}
	}
}

# Event rewards target the mechanic the country currently uses, including the legacy version.
gecko_add_prussian_militarization = {
	if = {
		limit = { has_government_mechanic = prussian_militarization_mechanic_3 }
		add_government_power = { mechanic_type = prussian_militarization_mechanic_3 power_type = prussian_militarized_society_3 value = $amount$ }
	}
	else_if = {
		limit = { has_government_mechanic = prussian_militarization_mechanic_2 }
		add_government_power = { mechanic_type = prussian_militarization_mechanic_2 power_type = prussian_militarized_society_2 value = $amount$ }
	}
	else_if = {
		limit = { has_government_mechanic = prussian_militarization_mechanic_1 }
		add_government_power = { mechanic_type = prussian_militarization_mechanic_1 power_type = prussian_militarized_society_1 value = $amount$ }
	}
	else_if = {
		limit = { has_government_mechanic = militarization_mechanic }
		add_government_power = { mechanic_type = militarization_mechanic power_type = militarized_society value = $amount$ }
	}
}
