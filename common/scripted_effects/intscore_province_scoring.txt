intscore_score_single_province = {
	if = {
		limit = {
			$PROVINCE$ = {
				OR = {
					owned_by = ROOT

					owner = {
						is_subject_of = ROOT
					}
				}
			}
		}

		# Parte dal valore base della provincia.
		set_variable = {
			which = intscore_points_to_add
			value = $BASE$
		}

		# Applica il moltiplicatore dell'epoca.
		multiply_variable = {
			which = intscore_points_to_add
			value = $MULTIPLIER$
		}

		# Aggiunge il risultato a IntScore.
		change_variable = {
			which = IntScore
			which = intscore_points_to_add
		}

		# Salva il risultato come YearlyIntScore.

		change_variable = {
			which = YearlyIntScore
			which = intscore_points_to_add
		}
	}
}
