intscore_clear_rank_slot = {
	if = {
		limit = {
			has_saved_global_event_target = $TARGET$
		}

		clear_global_event_target = $TARGET$
	}

	clr_global_flag = $FLAG$
}
intscore_select_rank_slot = {
	if = {
		limit = { check_variable = { which = intscore_remaining_candidates value = 1 } }
		random_country = {
			limit = {
				has_country_flag = intscore_participant
				check_variable = { which = IntScore value = 0.001 }
				NOT = { has_country_flag = intscore_ranking_selected }
			}
			save_global_event_target_as = $TARGET$
		}
		every_country = {
			limit = {
				has_country_flag = intscore_participant
				check_variable = { which = IntScore value = 0.001 }
				NOT = { has_country_flag = intscore_ranking_selected }
			}
			if = {
				limit = {
					event_target:$TARGET$ = { intscore_prefer_previous_candidate = yes }
				}
				save_global_event_target_as = $TARGET$
			}
		}
		event_target:$TARGET$ = {
			set_country_flag = intscore_ranking_selected
			set_variable = { which = intscore_current_rank value = $RANK$ }
		}
		set_global_flag = $FLAG$
		subtract_variable = { which = intscore_remaining_candidates value = 1 }
	}
}

intscore_assign_place_flags = {

	every_country = {
		limit = {
			OR = {
				has_country_flag = intscore_first_place
				has_country_flag = intscore_second_place
				has_country_flag = intscore_third_place
				has_country_flag = intscore_contender
			}
		}

		clr_country_flag = intscore_first_place
		clr_country_flag = intscore_second_place
		clr_country_flag = intscore_third_place
		clr_country_flag = intscore_contender
	}

	# Nuovo primo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_1_exists
		}

		event_target:intscore_rank_1 = {
			set_country_flag = intscore_first_place
		}
	}

	# Nuovo secondo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_2_exists
		}

		event_target:intscore_rank_2 = {
			set_country_flag = intscore_second_place
		}
	}

	# Nuovo terzo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_3_exists
		}

		event_target:intscore_rank_3 = {
			set_country_flag = intscore_third_place
		}
	}

	# Contender: quarto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_4_exists
		}

		event_target:intscore_rank_4 = {
			set_country_flag = intscore_contender
		}
	}

	# Contender: quinto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_5_exists
		}

		event_target:intscore_rank_5 = {
			set_country_flag = intscore_contender
		}
	}

	# Contender: sesto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_6_exists
		}

		event_target:intscore_rank_6 = {
			set_country_flag = intscore_contender
		}
	}
}
