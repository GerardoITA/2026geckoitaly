intscore_clear_rank_slot = {
	if = {
		limit = {
			has_saved_global_event_target = $TARGET$
		}

		clear_global_event_target = $TARGET$
	}

	clr_global_flag = $FLAG$
}
intscore_select_rank_slot = {
	if = {
		limit = {
			OR = {
				AND = {
					has_country_flag = intscore_participant

					check_variable = {
						which = IntScore
						value = 0.001
					}

					NOT = {
						has_country_flag = intscore_ranking_selected
					}
				}

				any_country = {
					has_country_flag = intscore_participant

					check_variable = {
						which = IntScore
						value = 0.001
					}

					NOT = {
						has_country_flag = intscore_ranking_selected
					}
				}
			}
		}

		# ROOT viene utilizzato come candidato iniziale se è eleggibile.
		if = {
			limit = {
				has_country_flag = intscore_participant

				check_variable = {
					which = IntScore
					value = 0.001
				}

				NOT = {
					has_country_flag = intscore_ranking_selected
				}
			}

			save_global_event_target_as = $TARGET$
		}

		# Se ROOT non è eleggibile, seleziona un candidato iniziale
		# tra i partecipanti non ancora classificati.
		else = {
			random_country = {
				limit = {
					has_country_flag = intscore_participant

					check_variable = {
						which = IntScore
						value = 0.001
					}

					NOT = {
						has_country_flag = intscore_ranking_selected
					}
				}

				save_global_event_target_as = $TARGET$
			}
		}

		# Esamina tutti i partecipanti non ancora classificati.
		every_country = {
			limit = {
				has_country_flag = intscore_participant

				check_variable = {
					which = IntScore
					value = 0.001
				}

				NOT = {
					has_country_flag = intscore_ranking_selected
				}
			}

			# Scope corrente: paese analizzato.
			#
			# Dentro event_target:$TARGET$:
			# - scope corrente = candidato massimo attuale;
			# - PREV = paese analizzato.
			#
			# Il controllo interno verifica:
			# candidato.IntScore >= paese_analizzato.IntScore.
			#
			# Il NOT rende vero il limite soltanto quando:
			# paese_analizzato.IntScore > candidato.IntScore.
			if = {
				limit = {
					event_target:$TARGET$ = {
						NOT = {
							check_variable = {
								which = IntScore
								which = PREV
							}
						}
					}
				}

				# Il paese analizzato ha un punteggio superiore
				# e diventa il nuovo candidato massimo.
				save_global_event_target_as = $TARGET$
			}
		}

		# Assegna definitivamente lo slot al massimo trovato
		# e lo esclude dalle selezioni successive.
		event_target:$TARGET$ = {
			set_country_flag = intscore_ranking_selected
		}

		set_global_flag = $FLAG$
	}
}

intscore_assign_place_flags = {

	# Pulisce eventuali piazzamenti precedenti anche da ROOT.
	clr_country_flag = intscore_first_place
	clr_country_flag = intscore_second_place
	clr_country_flag = intscore_third_place
	clr_country_flag = intscore_contender

	every_country = {
		limit = {
			OR = {
				has_country_flag = intscore_first_place
				has_country_flag = intscore_second_place
				has_country_flag = intscore_third_place
				has_country_flag = intscore_contender
			}
		}

		clr_country_flag = intscore_first_place
		clr_country_flag = intscore_second_place
		clr_country_flag = intscore_third_place
		clr_country_flag = intscore_contender
	}

	# Nuovo primo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_1_exists
		}

		event_target:intscore_rank_1 = {
			set_country_flag = intscore_first_place
		}
	}

	# Nuovo secondo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_2_exists
		}

		event_target:intscore_rank_2 = {
			set_country_flag = intscore_second_place
		}
	}

	# Nuovo terzo posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_3_exists
		}

		event_target:intscore_rank_3 = {
			set_country_flag = intscore_third_place
		}
	}

	# Contender: quarto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_4_exists
		}

		event_target:intscore_rank_4 = {
			set_country_flag = intscore_contender
		}
	}

	# Contender: quinto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_5_exists
		}

		event_target:intscore_rank_5 = {
			set_country_flag = intscore_contender
		}
	}

	# Contender: sesto posto.
	if = {
		limit = {
			has_global_flag = intscore_rank_6_exists
		}

		event_target:intscore_rank_6 = {
			set_country_flag = intscore_contender
		}
	}
}
