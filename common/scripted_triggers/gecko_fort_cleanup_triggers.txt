# Province scope. Only forts owned and controlled by the country being checked count.
# Capitals, occupied or besieged forts and recently acquired forts cannot form an illegal pair.
# Cleanup checks completed fort buildings; construction has a separate restriction below.
gecko_fort_cleanup_eligible = {
	owned_by = event_target:gecko_fort_cleanup_owner
	controlled_by = owner
	has_siege = no
	has_fort_building_trigger = yes
	is_capital = no
	NOT = {
		AND = {
			has_province_flag = PR_new_conquest_fort_delay
			NOT = { had_province_flag = { flag = PR_new_conquest_fort_delay days = 182 } }
		}
	}
}

# Province scope. The country effect saves the owner before entering provinces.
gecko_fort_cleanup_conflict = {
	gecko_fort_cleanup_eligible = yes
	any_neighbor_province = {
		owned_by = event_target:gecko_fort_cleanup_owner
		gecko_fort_cleanup_eligible = yes
	}
}

# Province scope. Upgrades and replacements beside unoccupied forts remain allowed.
# PREV inside any_neighbor_province is the construction province, not the builder.
gecko_fort_construction_allowed = {
	custom_trigger_tooltip = {
		tooltip = gecko_fort_construction_allowed_tt
		OR = {
			has_fort_building_trigger = yes
			NOT = {
				any_neighbor_province = {
					owned_by = PREV
					has_fort_building_trigger = yes
					OR = {
						NOT = { controlled_by = owner }
						has_siege = yes
					}
				}
			}
		}
	}
}
