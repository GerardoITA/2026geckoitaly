country_event = {
	id = intscore.999
	title = intscore.999.t
	desc = intscore.999.d
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		was_player = yes
		is_colonial_nation = no

		NOT = {
			has_country_flag = score_event_fired
		}

		is_month = 0
		NOT = {
			is_month = 1
		}

		is_year = 1444
		NOT = {
			is_year = 1500
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		# Calcola i punti delle province valide.
		#
		# Ogni provincia assegna:
		# BASE della provincia × MULTIPLIER dell'epoca.
		#
		# Per il periodo 1444-1499 il moltiplicatore è 1.

		set_variable = {
			which = YearlyIntScore
			value = 0
		}

		intscore_score_valid_provinces = {
			MULTIPLIER = 1
		}

		set_country_flag = intscore_participant
		set_country_flag = score_event_fired
		clr_global_flag = intscore_ranking_updated
	}

	option = {
		name = score_option_a

		ai_chance = {
			factor = 100
		}
	}
}

####### RICALCOLA CLASSIFICA 
country_event = {
	id = intscore.1060
	title = intscore.1060.t
	desc = intscore.1060.d
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		has_country_flag = intscore_participant
		has_country_flag = score_event_fired

		# Esclusivamente febbraio
		is_month = 1
		NOT = {
			is_month = 2
		}

		NOT = {
			has_global_flag = intscore_ranking_updated
		}

		NOT = {
			any_country = {
				has_country_flag = intscore_participant

				NOT = {
					has_country_flag = score_event_fired
				}
			}
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		set_global_flag = intscore_ranking_updated

		intscore_build_ranking = yes
		intscore_assign_place_flags = yes

		clr_country_flag = score_event_fired

		every_country = {
			limit = {
				has_country_flag = intscore_participant
			}

			clr_country_flag = score_event_fired
		}
	}

	option = {
		name = score_option_a
		ai_chance = {
			factor = 100
		}
	}
}
